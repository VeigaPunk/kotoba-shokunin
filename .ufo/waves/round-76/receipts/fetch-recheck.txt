## Reproducibility — the exact recheck script (was scratch-only; reproduced here so the run is repeatable)

```python
#!/usr/bin/env python3
"""Independent second fetch of three agentskills.io pages using python urllib
(different client stack from the wave's curl fetch). Records HTTP status,
byte length and sha256 of the raw response body."""
import hashlib
import json
import sys
import urllib.request
import urllib.error
from datetime import datetime, timezone

PAGES = [
    ("optimizing-descriptions", "https://agentskills.io/skill-creation/optimizing-descriptions.md"),
    ("evaluating-skills", "https://agentskills.io/skill-creation/evaluating-skills.md"),
    ("best-practices", "https://agentskills.io/skill-creation/best-practices.md"),
]

UA = "kotoba-r76-B26FetchRecheck/1.0 (urllib; independent recheck)"

def fetch(url):
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    try:
        with urllib.request.urlopen(req, timeout=60) as resp:
            body = resp.read()
            return resp.status, dict(resp.headers), body
    except urllib.error.HTTPError as e:
        body = e.read() if hasattr(e, "read") else b""
        return e.code, dict(e.headers or {}), body

def main():
    out = []
    for name, url in PAGES:
        status, headers, body = fetch(url)
        rec = {
            "name": name,
            "url": url,
            "fetched_at_utc": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
            "http_status": status,
            "bytes": len(body),
            "sha256": hashlib.sha256(body).hexdigest(),
            "content_type": headers.get("Content-Type"),
            "etag": headers.get("ETag"),
            "last_modified": headers.get("Last-Modified"),
            "client": "python-urllib",
        }
        out.append(rec)
        print(json.dumps(rec), file=sys.stderr)
        with open(f"/fast/scratch/hyaku/kotoba-r76/B26FetchRecheck/{name}.body", "wb") as f:
            f.write(body)
    with open("/fast/scratch/hyaku/kotoba-r76/B26FetchRecheck/recheck.json", "w") as f:
        json.dump(out, f, indent=1)
    print(json.dumps(out, indent=1))

if __name__ == "__main__":
    main()
```

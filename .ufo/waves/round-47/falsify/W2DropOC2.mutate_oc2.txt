def window(buf, off, old_len, new_len, new_bytes=None):
    lo = max(0, off - 70)
    head = buf[lo:off].decode("utf-8", "replace")
    span = buf[off:off + old_len].decode("utf-8", "replace")
    tail = buf[off + old_len:off + old_len + 70].decode("utf-8", "replace")
    return head, span, tail


def apply_all(orig: bytes, mut: dict):
    buf = orig
    recs = []
    for old, new, count in mut["edits"]:
        n = buf.count(old)
        if n != count:
            raise SystemExit("%s: anchor count %d != %d for %r" % (mut["id"], n, count, old[:50]))
        off = buf.find(old)
        line = buf.count(b"\n", 0, off) + 1
        before = window(buf, off, len(old), len(new))
        buf = buf.replace(old, new, count)
        after_head = buf[max(0, off - 70):off].decode("utf-8", "replace")
        after_span = new.decode("utf-8", "replace")
        after_tail = buf[off + len(new):off + len(new) + 70].decode("utf-8", "replace")
        recs.append(dict(kind="edit", line=line, offset=off,
                         before=before, after=(after_head, after_span, after_tail),
                         old=old.decode("utf-8", "replace"), new=new.decode("utf-8", "replace")))
    for app in mut["appends"]:
        recs.append(dict(kind="eof-append", text=app.decode("utf-8", "replace")))
        buf = buf + app
    return buf, recs


def run_verifier(path: pathlib.Path):
    cmd = [PY, "verifiers/verify_surfaces.py", "--opencode-strings", str(path),
           "--omp-strings", str(OMP)]
    p = subprocess.run(cmd, cwd=str(CLONE), capture_output=True, text=True, timeout=900)
    return p.returncode, p.stdout, p.stderr


def classify(mut, code, stdout):
    oc = [ln for ln in stdout.splitlines() if "surfaces.opencode_extract" in ln]
    oc_line = oc[0] if oc else ""
    oc_fail = oc_line.startswith("FAIL surfaces.opencode_extract")
    names_ok = all(n in oc_line for n in mut["names"]) if mut["names"] else True
    named_rule = ("OC-" in oc_line) if oc_fail else None
    pred = mut["predicted"]
    if code == 1 and oc_fail and names_ok:
        obs = "FAIL-opencode_extract-named"
    elif code == 1 and oc_fail:
        obs = "FAIL-opencode_extract-names-incomplete"
    elif code == 1:
        obs = "FAIL-other-check-only"
    elif code == 0:
        obs = "PASS-exit0"
    else:
        obs = "EXIT-%d" % code
    if pred == "kill":
        verdict = "KILLED" if obs == "FAIL-opencode_extract-named" else "SURVIVED" if code == 0 else "NOT-KILLED-AS-PREDICTED"
    elif pred == "false_red":
        verdict = "FALSE-RED-CONFIRMED" if code == 1 and oc_fail else "ACCEPTED (too-wide)"
    elif mut["group"] in ("control", "equiv"):
        verdict = "ACCEPTED" if code == 0 else "FALSE-RED"
    else:
        verdict = "SURVIVED" if code == 0 else "NOT-SURVIVING"
    return dict(observed=obs, verdict=verdict, oc_line=oc_line[:1500],
                names_ok=names_ok, named_rule=named_rule)


def job(orig: bytes, mut: dict):
    buf, recs = apply_all(orig, mut)
    MUTDIR.mkdir(parents=True, exist_ok=True)
    path = MUTDIR / ("%s.strings" % mut["id"])
    path.write_bytes(buf)
    sha = hashlib.sha256(buf).hexdigest()
    try:
        code, out, err = run_verifier(path)
    finally:
        path.unlink(missing_ok=True)
    cls = classify(mut, code, out)
    return dict(id=mut["id"], group=mut["group"], predicted=mut["predicted"],
                expect_names=mut["names"], exit=code, mutant_sha256=sha,
                bytes=len(buf), edits=recs, stdout=out, stderr=err[-800:], **cls)


def main():
    orig = DUMP.read_bytes()
    orig_sha = hashlib.sha256(orig).hexdigest()
    if orig_sha != DUMP_SHA:
        raise SystemExit("frozen dump sha mismatch: %s" % orig_sha)
    if orig.count(CFG) != 1 or orig.count(HOME) != 1:
        raise SystemExit("anchor not single-occurrence in frozen dump")
    verifier = CLONE / "verifiers" / "verify_surfaces.py"
    ver_sha = hashlib.sha256(verifier.read_bytes()).hexdigest()
    with cf.ThreadPoolExecutor(max_workers=4) as ex:
        futs = {ex.submit(job, orig, m): m["id"] for m in M}
        results = {}
        for f in cf.as_completed(futs):
            results[futs[f]] = f.result()
    ordered = [results[m["id"]] for m in M]

    FALSIFY.mkdir(parents=True, exist_ok=True)
    header = dict(dump=str(DUMP), dump_sha256=orig_sha, omp_dump=str(OMP),
                  clone_head="7ad7fadfc09354b74fb66605dfb0c1182c115a2f",
                  verifier_sha256=ver_sha, n_mutants=len(ordered))
    (FALSIFY / (PREFIX + ".mutants.json")).write_text(
        json.dumps(dict(header=header, results=[{k: v for k, v in r.items() if k != "stdout"}
                                                 for r in ordered]), indent=1))
    log = []
    for r in ordered:
        log.append("==== %s  group=%s predicted=%s exit=%d verdict=%s\n%s%s\n" % (
            r["id"], r["group"], r["predicted"], r["exit"], r["verdict"], r["stdout"],
            ("STDERR: " + r["stderr"]) if r["stderr"].strip() else ""))
    (FALSIFY / (PREFIX + ".run.log")).write_text("".join(log))
    patch = []
    for r in ordered:
        patch.append("## %s  (group=%s; predicted=%s; exit=%d; %s)\n" % (
            r["id"], r["group"], r["predicted"], r["exit"], r["verdict"]))
        for e in r["edits"]:
            if e["kind"] == "edit":
                b = e["before"]
                a = e["after"]
                patch.append("@@ dump line %d, offset %d @@\n- %s[[%s]]%s\n+ %s[[%s]]%s\n" % (
                    e["line"], e["offset"], b[0], b[1], b[2], a[0], a[1], a[2]))
            else:
                patch.append("@@ EOF append @@\n+ %r\n" % e["text"])
        patch.append("\n")
    (FALSIFY / (PREFIX + ".patch")).write_text("".join(patch))

    print("%-36s %-14s %-4s %-30s %s" % ("id", "group", "exit", "observed", "verdict"))
    for r in ordered:
        print("%-36s %-14s %-4d %-30s %s" % (r["id"], r["group"], r["exit"], r["observed"], r["verdict"]))
    print("dump sha ok:", orig_sha == DUMP_SHA, " verifier sha:", ver_sha[:16])


if __name__ == "__main__":
    main()

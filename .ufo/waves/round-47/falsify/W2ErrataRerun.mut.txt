#!/usr/bin/env python3
"""Mutation harness for the round-47 C3 GOVERNANCE_ERRATA row.

Runs the live governance check (verifiers/verify_skill_checklist.py) against
mutated copies of the verifier / LOG.md / worktree and records the observed
status.  Each mutant must FAIL (or otherwise change the outcome) for the
erratum to be load-bearing; a mutant that still PASSes is a hole.
"""
import hashlib
import json
import pathlib
import re
import shutil
import subprocess
import sys
import tempfile

REPO = pathlib.Path(sys.argv[1]).resolve()
VER = REPO / "verifiers" / "verify_skill_checklist.py"
LOG = REPO / "LOG.md"

results = []


def run_check(repo):
    p = subprocess.run(
        [sys.executable, "-B", "verifiers/verify_skill_checklist.py", "--json",
         "--check", "governance.citation_resolves"],
        cwd=repo, capture_output=True, text=True)
    try:
        d = json.loads(p.stdout)
    except Exception:
        return p.returncode, "UNPARSEABLE", p.stdout[:200] + p.stderr[:200]
    c = d["checks"][0]
    return p.returncode, c["status"], c["detail"]


def restore():
    shutil.copy2(VER.with_suffix(".py.bak"), VER)
    shutil.copy2(LOG.with_suffix(".md.bak"), LOG)
    for extra in (REPO / ".ufo/round-45/drafts/v10/post-r44-hunt.py",):
        if extra.exists():
            extra.unlink()


VER.with_suffix(".py.bak").write_bytes(VER.read_bytes())
LOG.with_suffix(".md.bak").write_bytes(LOG.read_bytes())

base = run_check(REPO)
results.append(("BASELINE (unmutated)", base[0], base[1], base[2]))

src = VER.read_text(encoding="utf-8")


def mutant(name, transform, repo=REPO, pre=None, allow_same=False):
    restore()
    new = transform(src)
    assert new != src or allow_same, (
        "mutant %s did not change the file" % name)
    VER.write_text(new, encoding="utf-8")
    if pre:
        pre()
    out = run_check(repo)
    results.append((name, out[0], out[1], out[2]))
    restore()


# M1: horizon moved below the pinned line -> row is past horizon
mutant("M1 horizon 2315 -> 2110",
       lambda s: s.replace("GOVERNANCE_ERRATA_HORIZON = 2315",
                           "GOVERNANCE_ERRATA_HORIZON = 2110"))

# M2: line digest falsified
mutant("M2 digest falsified",
       lambda s: s.replace(
           "7dcbf123313d46c469cab9f5dac2daa7567e54ac1ddf7cffdd1fd38c74200135",
           "0" * 64))

# M3: absence wording falsified
mutant("M3 absence wording falsified",
       lambda s: s.replace(
           "each exited 2 because the referenced runners are absent",
           "each exited 2 because the referenced runners are PRESENT"))

# M4: pinned line number shifted off the real line
mutant("M4 pinned line 2111 -> 2112",
       lambda s: s.replace("    (2111,\n", "    (2112,\n"))

# M5: pinned token replaced by another (non-subject) token
mutant("M5 pinned token -> corpus/nope.md",
       lambda s: s.replace('".ufo/round-45/drafts/v10/post-r44-hunt.py"',
                           '"corpus/nope.md"'))

# M6: the row is deleted entirely -> the dead subject must surface
mutant("M6 row removed (errata=())",
       lambda s: s.replace(
           "GOVERNANCE_ERRATA = (\n    (2111,\n"
           "     \"7dcbf123313d46c469cab9f5dac2daa7567e54ac1ddf7cffdd1fd38c74200135\",\n"
           "     \".ufo/round-45/drafts/v10/post-r44-hunt.py\",\n"
           "     \"each exited 2 because the referenced runners are absent\"),\n)",
           "GOVERNANCE_ERRATA = ()"))

# M7: the excused path is made to exist -> the erratum goes stale
def _mkpath():
    p = REPO / ".ufo/round-45/drafts/v10/post-r44-hunt.py"
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text("# planted\n", encoding="utf-8")


mutant("M7 excused path created (stale erratum)", lambda s: s, pre=_mkpath,
       allow_same=True)

# M8: LOG.md gains a line above 2111 -> the line pin must drift-fail
def _insert_head():
    lines = LOG.read_text(encoding="utf-8").splitlines(keepends=True)
    LOG.write_text("- planted line\n" + "".join(lines), encoding="utf-8")


restore()
_insert_head()
out = run_check(REPO)
results.append(("M8 LOG.md line inserted above 2111 (pin drift)", out[0], out[1],
                out[2]))
restore()

# M9: the LOG line 2111 text is edited -> the pin must drift-fail
def _edit_2111():
    lines = LOG.read_text(encoding="utf-8").splitlines(keepends=True)
    lines[2110] = lines[2110].replace("runners are absent",
                                       "runners were absent")
    LOG.write_text("".join(lines), encoding="utf-8")


restore()
_edit_2111()
out = run_check(REPO)
results.append(("M9 LOG.md:2111 text edited (pin drift)", out[0], out[1], out[2]))
restore()

print(json.dumps(results, indent=1))

#!/usr/bin/env python3
"""Independent audit of the round-47 landing's governance citations.

Re-implements the documented governance.citation_resolves subject rule from
the verifier's own docstring (verifiers/verify_skill_checklist.py:187-197)
WITHOUT importing the verifier, so the finding does not depend on the tool
under test.  Focus: the newly appended LOG.md tail and the SCORECARD.md
edits of 7ad7fad -- every repo-rooted backtick token introduced there must
resolve at HEAD, and none may hide behind the quoted/schematic exclusions.
"""
import json
import pathlib
import re
import subprocess
import sys

REPO = pathlib.Path(sys.argv[1]).resolve()
BASE = sys.argv[2] if len(sys.argv) > 2 else "c61a973"
TIP = sys.argv[3] if len(sys.argv) > 3 else "7ad7fad"

ROOTS = ("corpus/", "verifiers/", "opencode/", ".ufo/", "../")
TRAILING = ".,;:!?)]}\"'"
SCHEMATIC = re.compile(r"(?:^|/)round-N(?:/|\.|$)")
SPAN = re.compile(r"`([^`\n]+)`")


def git_show(rev, path):
    return subprocess.run(["git", "show", f"{rev}:{path}"], cwd=REPO,
                          capture_output=True, text=True, check=True).stdout


def strip_dressing(tok):
    tok = tok.rstrip(TRAILING)
    m = re.match(r"^(.+\.\w+):\d+(-\d+)?$", tok)
    return m.group(1) if m else tok


def tokens(text, lineno_offset=1):
    out = []
    for lineno, line in enumerate(text.splitlines(), lineno_offset):
        for span in SPAN.finditer(line):
            for word in re.finditer(r"\S+", span.group(1)):
                tok = word.group(0).split("#", 1)[0]
                if not tok or any(c in tok for c in "<>*?$"):
                    continue
                if tok.startswith(("http", "~", "-")):
                    continue
                norm = strip_dressing(tok)
                if norm.startswith("./"):
                    norm = norm[2:]
                if not norm.startswith(ROOTS):
                    continue
                if norm.rstrip("/") and "/" not in norm.rstrip("/"):
                    continue
                out.append((lineno, norm, line))
    return out


def resolve(tok):
    parts = tok.split("/")
    if ".." not in parts:
        return (REPO / tok).exists()
    hops = 0
    while parts and parts[0] == "..":
        parts.pop(0)
        hops += 1
    if hops > 3 or ".." in parts or not parts:
        return False
    base = REPO
    for _ in range(hops):
        base = base.parent
    return (base / "/".join(parts)).exists()


def quoted_or_schematic(norm):
    return ("…" in norm or "..." in norm or bool(SCHEMATIC.search(norm)))


report = {"repo": str(REPO), "base": BASE, "tip": TIP, "files": {}}

for name in ("LOG.md", "SCORECARD.md"):
    base_tok = tokens(git_show(BASE, name))
    tip_tok = tokens(git_show(TIP, name))
    base_set = {(t[1]) for t in base_tok}
    new = [t for t in tip_tok if t[1] not in base_set]
    # multiplicity-aware: a token repeated in the tip but absent from base
    dead, live, excl = [], [], []
    for lineno, tok, line in new:
        if quoted_or_schematic(tok):
            excl.append((lineno, tok))
        elif resolve(tok):
            live.append((lineno, tok))
        else:
            dead.append((lineno, tok, line.strip()[:160]))
    report["files"][name] = {
        "base_subjects": len(base_tok),
        "tip_subjects": len(tip_tok),
        "new_distinct_subjects": len(new),
        "new_live": len(live),
        "new_schematic_or_ellipsis": [f"{n}:{t}" for n, t in excl],
        "new_dead": dead,
    }

print(json.dumps(report, indent=1))

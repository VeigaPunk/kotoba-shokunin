def _parse_block(text):
    out = {}
    for m in re.finditer(r"^- ([^:\n]+): (.*)$", text, re.M):
        out.setdefault(m.group(1), m.group(2))
    return out
def _strings(o):
    if isinstance(o, str):
        yield o
    elif isinstance(o, dict):
        for v in o.values():
            yield from _strings(v)
    elif isinstance(o, list):
        for v in o:
            yield from _strings(v)
def _leak():
